Android Kotlin Quickstart (`ShrotaStreamingEngine.kt`)

val engine = ShrotaStreamingEngine(
    encoderModelPath = "${filesDir}/shrota_tiny_encoder_int8.onnx",
    shuddhaModelPath = "${filesDir}/shrota_shuddha_int8.onnx",
    vocab = vocabList,
    dModel = 128,
    nLayers = 8
)

engine.startLiveMicrophone(
    listener = object : ShrotaStreamingEngine.Listener {
        override fun onPartialTranscript(text: String, ms: Float, rtf: Float) {
            liveSubtitle.text = text // 160ms low-latency preview
        }
        override fun onFinalClause(raw: String, shuddha: String, conf: Float, ms: Float) {
            editor.append("$shuddha ") // Hraswa-Dirgha & Punctuation polished!
        }
    },
    extractMel16x80 = { pcm2560 -> melExtractor.compute16x80(pcm2560) }
)
